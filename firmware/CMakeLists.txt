cmake_minimum_required(VERSION 3.20)
set(TUNER_TARGET "native" CACHE STRING "native or rp2040")
if(TUNER_TARGET STREQUAL "rp2040")
    if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
    endif()
    if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
        message(FATAL_ERROR "Set PICO_SDK_PATH to pinned Pico SDK 2.2.0")
    endif()
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(pianotuner_firmware LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(tuner_core core/core.cpp core/parser.cpp)
target_include_directories(tuner_core PUBLIC core PRIVATE vendor)
target_compile_options(tuner_core PRIVATE -Wall -Wextra -Wpedantic -fstack-usage)
if(TUNER_TARGET STREQUAL "native")
    add_executable(tuner_native native/main.cpp)
    target_link_libraries(tuner_native PRIVATE tuner_core)
    add_executable(tuner_allocation_check native/allocation_check.cpp)
    target_link_libraries(tuner_allocation_check PRIVATE tuner_core)
    add_executable(tuner_framing_check native/framing_check.cpp)
    target_link_libraries(tuner_framing_check PRIVATE tuner_core)
    if(MINGW)
        target_link_options(tuner_native PRIVATE -static)
        target_link_options(tuner_allocation_check PRIVATE -static)
        target_link_options(tuner_framing_check PRIVATE -static)
    endif()
    enable_testing()
    add_test(NAME allocation_free_core COMMAND tuner_allocation_check)
    add_test(NAME bounded_stream_framing COMMAND tuner_framing_check)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME protocol_conformance COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../tools/check_firmware_conformance.py $<TARGET_FILE:tuner_native>)
elseif(TUNER_TARGET STREQUAL "rp2040")
    pico_sdk_init()
    if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.2.0")
        message(FATAL_ERROR "Expected pinned Pico SDK 2.2.0")
    endif()
    add_executable(pianotuner_disabled rp2040/main.cpp)
    target_link_libraries(pianotuner_disabled PRIVATE tuner_core pico_stdlib hardware_watchdog)
    pico_enable_stdio_usb(pianotuner_disabled 1)
    pico_enable_stdio_uart(pianotuner_disabled 0)
    target_compile_definitions(pianotuner_disabled PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000 PICO_STACK_SIZE=65536)
    set_source_files_properties(rp2040/main.cpp PROPERTIES COMPILE_OPTIONS "-fstack-usage")
    # Fixed parser/request/reply buffers need more than the SDK's default 4 KiB
    # scratch-bank stack. Reserve the upper 64 KiB of main SRAM for core 0.
    file(READ "${PICO_SDK_PATH}/src/rp2_common/pico_crt0/rp2040/memmap_default.ld" TUNER_LINKER)
    string(REPLACE "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 256k"
        "RAM(rwx) : ORIGIN =  0x20000000, LENGTH = 192k\n    STACK(rwx) : ORIGIN = 0x20030000, LENGTH = 64k" TUNER_LINKER "${TUNER_LINKER}")
    string(REPLACE "KEEP(*(.stack*))\n    } > SCRATCH_Y" "KEEP(*(.stack*))\n    } > STACK" TUNER_LINKER "${TUNER_LINKER}")
    string(REPLACE "__StackTop = ORIGIN(SCRATCH_Y) + LENGTH(SCRATCH_Y);"
        "__StackTop = ORIGIN(STACK) + LENGTH(STACK);" TUNER_LINKER "${TUNER_LINKER}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/memmap_tuner.ld" "${TUNER_LINKER}")
    pico_set_linker_script(pianotuner_disabled "${CMAKE_CURRENT_BINARY_DIR}/memmap_tuner.ld")
    pico_add_extra_outputs(pianotuner_disabled)
else()
    message(FATAL_ERROR "Unknown TUNER_TARGET")
endif()
